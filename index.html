document.addEventListener('DOMContentLoaded', () => { // 1. НАСТРОЙКИ СЛОЕВ const LAYER_SETTINGS = { 'bg': { x: 0, y: 0, scale: 1.15, depth: -50, radius: 20, blur: 2, opacity: 1, clip: true }, 'chains-back': { x: 0, y: 0, scale: 1, depth: 20, radius: 20, blur: 0, opacity: 1, clip: true }, 'hand': { x: 0, y: 0, scale: 0.9, depth: 50, radius: 20, blur: 0, opacity: 1, clip: true }, 'card-frame': { x: 0, y: 0, scale: 1, depth: 80, radius: 20, blur: 0, opacity: 1, clip: true }, 'chains-front': { x: 0, y: 0, scale: 0.8, depth: 130, radius: 20, blur: 0, opacity: 1, clip: false } }; // 2. ПРИМЕНЕНИЕ НАСТРОЕК const wrappers = document.querySelectorAll('.layer-wrapper'); wrappers.forEach(wrapper => { const layerName = wrapper.dataset.layer; const settings = LAYER_SETTINGS[layerName]; if (!settings) return; const img = wrapper.querySelector('img'); wrapper.style.transform = `translate(${settings.x}%, ${settings.y}%) scale(${settings.scale})`; img.style.transform = `translateZ(${settings.depth}px)`; img.style.filter = `blur(${settings.blur}px)`; img.style.opacity = settings.opacity; if (settings.clip) { img.style.clipPath = `inset(0 round ${settings.radius}px)`; } else { img.style.clipPath = 'none'; } }); // 3. ЭФФЕКТ НАКЛОНА const scene = document.getElementById('scene'); const card = document.getElementById('card'); if (scene && card) { const MAX_ROTATE = 15; let currentX = 0, currentY = 0; let targetX = 0, targetY = 0; // --- ПК (Мышка) --- // Слушаем движение мышки по ВСЕМУ ОКНУ, чтобы увеличить радиус действия window.addEventListener('mousemove', (e) => { // Вычисляем позицию мышки относительно центра экрана const x = (e.clientX / window.innerWidth) - 0.5; const y = (e.clientY / window.innerHeight) - 0.5; targetX = y * -MAX_ROTATE; targetY = x * MAX_ROTATE; }); // --- ТЕЛЕФОН (Палец) --- // Слушаем касания по всей ширине окна, чтобы было удобно водить пальцем window.addEventListener('touchmove', (e) => { if (e.touches.length > 0) { const touch = e.touches[0]; const x = (touch.clientX / window.innerWidth) - 0.5; const y = (touch.clientY / window.innerHeight) - 0.5; targetX = y * -MAX_ROTATE; targetY = x * MAX_ROTATE; } }, { passive: true }); // Возврат в исходное положение, когда мышка или палец уходят window.addEventListener('mouseleave', () => { targetX = 0; targetY = 0; }); window.addEventListener('touchend', () => { targetX = 0; targetY = 0; }); // --- АНИМАЦИЯ (с инерцией) --- function animate() { // Плавно догоняем цель (0.1 - это скорость догоняния, можно менять) currentX += (targetX - currentX) * 0.1; currentY += (targetY - currentY) * 0.1; card.style.transform = `rotateX(${currentX}deg) rotateY(${currentY}deg)`; requestAnimationFrame(animate); } animate(); } });
